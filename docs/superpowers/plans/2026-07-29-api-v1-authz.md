# /api/v1 Authorization Hardening Implementation Plan (revised)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make every `/api/v1` JSON:API resource deny-by-default so a map API token can reach only its own map and resources linked to that map, without breaking internal or session callers.

**Architecture:** Add `Ash.Policy.Authorizer` + a `policies` block (a `bypass` for trusted actors, then action policies) to each route-exposed resource, gated by the domain's `authorize :when_requested`. A shared `WandererApp.Api.Policies.MapScoped` module supplies the `Trusted` bypass check, a filter check for reads, and write checks (direct-attribute or parent-query). The one hand-written v1 controller gets a direct guard.

**Tech Stack:** Elixir, Ash 3.9, AshJsonApi 1.4, AshPostgres, ExUnit.

## Global Constraints

- Ash `authorize` mode: `:when_requested`.
- **Three actor shapes:** API token `%ActorWithMap{map: %{id: _}}` → scoped; session `%ActorWithMap{map: nil}` → trusted; internal `%WandererApp.Api.User{}` or `%WandererApp.Api.Character{}` → trusted.
- **Policy composition is a `bypass` for trusted actors, THEN action policies.** Never two stacked unconditional `policy` blocks — Ash requires all applicable policies to pass, which would deny everyone.
- Reads use **filter** checks (out-of-scope → excluded → list subset / `GET :id` 404). Writes use checks that yield **403** out-of-scope.
- ACLs (`access_list`, `access_list_member`, `map_access_list`) are **read-only for token actors**; writes forbidden for tokens (allowed for trusted via bypass).
- `user_activity`: token actors **denied** (no single-map link).
- Token map extracted via `WandererApp.Api.ActorHelpers.get_map/1`.
- Tests: `use WandererAppWeb.ApiCase`, `import WandererAppWeb.Factory`, authenticate with `create_authenticated_conn(conn, map)` (uses `map.public_api_key`). Missing factories (`map_subscription`, `map_default_settings`, `user_activity`, `map_user_settings`) are created via `Ash.create/2` helpers in-test or added to the factory.
- No global domain bypass block. Every resource opts in explicitly.
- Before each commit: `mix compile --warnings-as-errors` and the task's tests.

## Route-exposed resource inventory (authoritative)

Writable/readable via `/api/v1`: `map`, `map_system`, `map_connection`, `map_subscription`, `map_user_settings`, `map_character_settings`, `map_access_list`, `map_default_settings`, `map_system_signature`, `map_system_structure`, `map_system_comment`, `access_list`, `access_list_member`, `user_activity`. Routes-block-but-no-operations (defense-in-depth only): `map_solar_system`, `map_state`, `ship_type_info`, `user`.

```elixir
# lib/wanderer_app/api/policies/map_scoped.ex
defmodule WandererApp.Api.Policies.MapScoped do
  @moduledoc """
  Policy checks scoping `/api/v1` requests to an `ActorWithMap` token's map.
  Trusted actors (plain User/Character, or a session ActorWithMap with nil map)
  are authorized via a bypass and never reach the scoped checks.
  """
  alias WandererApp.Api.ActorHelpers
  alias WandererApp.Api.ActorWithMap

  def trusted, do: {__MODULE__.Trusted, []}
  def in_token_map(path) when is_list(path), do: {__MODULE__.InTokenMap, path: path}
  def write_direct(attr \\ :map_id), do: {__MODULE__.WriteDirect, attr: attr}
  def parent_in_token_map(path) when is_list(path), do: {__MODULE__.ParentInTokenMap, path: path}

  def create_parent_in_token_map(parent_resource, fk),
    do: {__MODULE__.CreateParentInTokenMap, parent_resource: parent_resource, fk: fk}

  defmodule Trusted do
    use Ash.Policy.SimpleCheck
    alias WandererApp.Api.ActorWithMap

    @impl true
    def describe(_), do: "actor is trusted (internal User/Character or session actor without a map)"

    @impl true
    def match?(%ActorWithMap{map: nil}, _ctx, _opts), do: true
    def match?(%ActorWithMap{}, _ctx, _opts), do: false
    def match?(%WandererApp.Api.User{}, _ctx, _opts), do: true
    def match?(%WandererApp.Api.Character{}, _ctx, _opts), do: true
    def match?(_actor, _ctx, _opts), do: false
  end

  defmodule InTokenMap do
    use Ash.Policy.FilterCheck
    alias WandererApp.Api.ActorHelpers

    @impl true
    def describe(_), do: "row belongs to the token's map"

    @impl true
    def filter(actor, _ctx, opts) do
      path = Keyword.fetch!(opts, :path)

      case ActorHelpers.get_map(%{actor: actor}) do
        %{id: map_id} -> ref_eq(path, map_id)
        _ -> expr(false)
      end
    end

    defp ref_eq([attr], v), do: expr(^Ash.Expr.ref(attr) == ^v)

    defp ref_eq(path, v) do
      {rel, [attr]} = Enum.split(path, -1)
      expr(^Ash.Expr.ref(attr, rel) == ^v)
    end
  end

  # Update/Destroy for relationship-path resources: same filter semantics.
  defmodule ParentInTokenMap do
    use Ash.Policy.FilterCheck
    alias WandererApp.Api.ActorHelpers

    @impl true
    def describe(_), do: "row's parent belongs to the token's map"

    @impl true
    def filter(actor, _ctx, opts) do
      path = Keyword.fetch!(opts, :path)

      case ActorHelpers.get_map(%{actor: actor}) do
        %{id: map_id} ->
          {rel, [attr]} = Enum.split(path, -1)
          expr(^Ash.Expr.ref(attr, rel) == ^map_id)

        _ ->
          expr(false)
      end
    end
  end

  defmodule WriteDirect do
    use Ash.Policy.SimpleCheck
    alias WandererApp.Api.ActorHelpers

    @impl true
    def describe(_), do: "changeset target belongs to the token's map"

    @impl true
    def match?(actor, %{changeset: %Ash.Changeset{} = cs}, opts) do
      attr = Keyword.get(opts, :attr, :map_id)

      with %{id: map_id} <- ActorHelpers.get_map(%{actor: actor}),
           row when not is_nil(row) <-
             Ash.Changeset.get_attribute(cs, attr) || Map.get(cs.data || %{}, attr) do
        row == map_id
      else
        _ -> false
      end
    end

    def match?(_actor, _ctx, _opts), do: false
  end

  # Create for relationship-path resources: verify the parent row is in the token map
  # via an authorization query (relationship data is not loaded on create).
  defmodule CreateParentInTokenMap do
    use Ash.Policy.SimpleCheck
    alias WandererApp.Api.ActorHelpers

    @impl true
    def describe(_), do: "created row's parent belongs to the token's map"

    @impl true
    def match?(actor, %{changeset: %Ash.Changeset{} = cs}, opts) do
      parent = Keyword.fetch!(opts, :parent_resource)
      fk = Keyword.fetch!(opts, :fk)

      with %{id: map_id} <- ActorHelpers.get_map(%{actor: actor}),
           parent_id when not is_nil(parent_id) <- Ash.Changeset.get_attribute(cs, fk) do
        case Ash.read_one(
               Ash.Query.filter(parent, id == ^parent_id and map_id == ^map_id),
               authorize?: false
             ) do
          {:ok, nil} -> false
          {:ok, _row} -> true
          _ -> false
        end
      else
        _ -> false
      end
    end

    def match?(_actor, _ctx, _opts), do: false
  end
end
```
